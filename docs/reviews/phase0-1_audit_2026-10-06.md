### ⚠️ P3 — Nên sửa (Clean architecture)

**lib/socket.ts — đổi sang lazy init**

```typescript
// Hiện tại (module-scope — không lazy):
export const socket: Socket = io(API_URL, { ... });

// Nên đổi thành:
let _socket: Socket | null = null;
export function getSocket(): Socket {
  if (!_socket) {
    _socket = io(API_URL, { ... });
  }
```
